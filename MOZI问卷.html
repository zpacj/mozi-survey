function showToast(msg, type = '') {
  const t = document.getElementById('toast');
  t.textContent = msg;
  t.className = 'toast show ' + type;
  setTimeout(() => t.className = 'toast', 2500);
}

function addRow(listId, name, label, firstPlaceholder) {
  const list = document.getElementById(listId);
  const idx = list.children.length + 1;
  const placeholder = firstPlaceholder || (idx === 1 ? `${label}1` : `${label}${idx}`);
  const row = document.createElement('div');
  row.className = 'job-row';
  row.innerHTML = `
    
    
  `;
  list.appendChild(row);
  updateRemoveButtons(listId);
  row.querySelector('input').focus();
}

function removeJobRow(btn) {
  removeRow(btn, 'q1_jobs_list');
}

function removeRow(btn, listId) {
  const row = btn.closest('.job-row');
  if (!row) return;
  const list = document.getElementById(listId);
  if (list.children.length <= 1) return;
  row.remove();
  refreshRowPlaceholders(listId);
  updateRemoveButtons(listId);
}

function refreshRowPlaceholders(listId) {
  const list = document.getElementById(listId);
  if (!list) return;
  Array.from(list.children).forEach((row, i) => {
    const input = row.querySelector('input');
    if (input && !input.value.trim()) {
      const ph = input.placeholder;
      // 只刷新纯 "岗位N/依据N/方式N..." 这类纯数字占位符
      const m = ph.match(/^(岗位|依据|方式|能力|成果|背书方|角色|议题)(\d+.*)$/);
      if (m) {
        input.placeholder = `${m[1]}${i + 1}${i === 0 && m[1] === '岗位' ? '（如：机械工程师）' : ''}`;
      }
    }
  });
}

function updateRemoveButtons(listId) {
  if (listId) {
    const list = document.getElementById(listId);
    if (!list) return;
    const rows = list.querySelectorAll('.job-row');
    rows.forEach((row) => {
      const btn = row.querySelector('.row-remove');
      if (btn) {
        btn.disabled = rows.length === 1;
        btn.style.opacity = rows.length === 1 ? '0.3' : '1';
        btn.style.cursor = rows.length === 1 ? 'not-allowed' : 'pointer';
      }
    });
    return;
  }
  // 不传 listId 时，刷新所有 .list-input
  document.querySelectorAll('.list-input').forEach(list => {
    const rows = list.querySelectorAll('.job-row');
    rows.forEach((row) => {
      const btn = row.querySelector('.row-remove');
      if (btn) {
        btn.disabled = rows.length === 1;
        btn.style.opacity = rows.length === 1 ? '0.3' : '1';
        btn.style.cursor = rows.length === 1 ? 'not-allowed' : 'pointer';
      }
    });
  });
}

function updateProgress() {
  const required = document.querySelectorAll('[required]');
  let filled = 0, total = required.length;
  radioChecked = {};

  required.forEach(el => {
    if (el.type === 'radio') {
      const name = el.name;
      if (document.querySelector(`input[name="${name}"]:checked`)) {
        if (!radioChecked[name]) {
          filled++;
          radioChecked[name] = true;
        }
      }
    } else if (el.type === 'checkbox') {
      if (el.checked) filled++;
    } else if (el.value.trim()) {
      filled++;
    }
  });

  const percent = total > 0 ? Math.round((filled / total) * 100) : 0;
  document.getElementById('progressText').textContent = `已填写 ${filled} / ${total}`;
  document.getElementById('progressPercent').textContent = `${percent}%`;
  document.getElementById('progressFill').style.width = `${percent}%`;
}

document.addEventListener('change', (e) => {
  const target = e.target;

  if (target.matches('.option input[type="radio"], .option input[type="checkbox"]')) {
    const option = target.closest('.option');
    if (option) option.classList.add('checked');

    if (target.type === 'radio') {
      const name = target.name;
      document.querySelectorAll(`input[name="${name}"]`).forEach(r => {
        const opt = r.closest('.option');
        if (opt && r !== target) opt.classList.remove('checked');
      });
    }

    if (target.dataset.other !== undefined) {
      const container = target.closest('.options');
      const otherInput = container.querySelector('.option-other');
      if (target.checked) otherInput.classList.add('visible');
      else otherInput.classList.remove('visible');
    }
  }

  updateProgress();
});

document.addEventListener('input', updateProgress);

function collectData() {
  const form = document.getElementById('surveyForm');
  const fd = new FormData(form);
  const data = {};

  for (let [key, value] of fd.entries()) {
    if (data[key] !== undefined) {
      if (!Array.isArray(data[key])) data[key] = [data[key]];
      data[key].push(value);
    } else {
      data[key] = value;
    }
  }

  document.querySelectorAll('input[type="radio"]').forEach(r => {
    if (data[r.name] === undefined) data[r.name] = null;
  });

  if (data.q1_jobs) {
    data.q1_jobs = Array.isArray(data.q1_jobs) ? data.q1_jobs : [data.q1_jobs];
    data.q1_jobs = data.q1_jobs.filter(v => v && v.trim());
  }

  data._submitted_at = new Date().toISOString();
  data._form_version = '1.0';

  return data;
}

document.getElementById('surveyForm').addEventListener('submit', (e) => {
  e.preventDefault();

  const missing = [];
  const seenRadios = new Set();
  document.querySelectorAll('[required]').forEach(el => {
    if (el.type === 'radio') {
      if (!seenRadios.has(el.name)) {
        seenRadios.add(el.name);
        if (!document.querySelector(`input[name="${el.name}"]:checked`)) {
          missing.push(el.name);
        }
      }
    } else if (el.type === 'checkbox') {
      if (!el.checked) missing.push(el.name);
    } else if (!el.value.trim()) {
      missing.push(el.name);
    }
  });

  if (missing.length > 0) {
    showToast(`还有 ${missing.length} 项必填未完成`, 'error');
    const firstMissing = document.querySelector(`[name="${missing[0]}"]`);
    if (firstMissing) {
      firstMissing.scrollIntoView({ behavior: 'smooth', block: 'center' });
      firstMissing.focus();
    }
    return;
  }

  formData = collectData();
  // 收集必填字段名（供后端计算完成率）
  const requiredKeys = [...seenRadios];
  document.querySelectorAll('[required]').forEach(el => {
    if (!requiredKeys.includes(el.name) && el.type !== 'radio') {
      requiredKeys.push(el.name);
    }
  });
  formData._required_keys = requiredKeys;

  // 提交到后端
  const submitBtn = e.target.querySelector('button[type="submit"]') || document.querySelector('button[form="surveyForm"]');
  const oldText = submitBtn ? submitBtn.textContent : '';
  if (submitBtn) { submitBtn.disabled = true; submitBtn.textContent = '⏳ 提交中…'; }

  fetch(API_BASE + '/api/surveys', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify(formData)
  })
  .then(r => r.json())
  .then(j => {
    if (j.ok) {
      showModal(`✅ 提交成功！\n\n编号：${j.interview_no}\n完成率：${j.completion_rate}%\n\n已保存到云端，可在「问卷管理」标签查看。`);
    } else {
      throw new Error(j.error || '提交失败');
    }
  })
  .catch(err => {
    showModal(`⚠️ 自动保存到云端失败：${err.message}\n\n您仍可点击「导出 JSON」手动备份数据。`);
  })
  .finally(() => {
    if (submitBtn) { submitBtn.disabled = false; submitBtn.textContent = oldText || '✅ 提交问卷'; }
  });
});

function previewData() {
  formData = collectData();
  document.getElementById('previewContent').textContent = JSON.stringify(formData, null, 2);
  document.getElementById('previewModal').classList.add('show');
}

function showModal(msg) {
  document.getElementById('previewContent').textContent = msg;
  document.getElementById('previewModal').classList.add('show');
}

function closeModal() {
  document.getElementById('previewModal').classList.remove('show');
}

function exportJSON() {
  formData = collectData();
  const blob = new Blob([JSON.stringify(formData, null, 2)], { type: 'application/json' });
  const url = URL.createObjectURL(blob);
  const a = document.createElement('a');
  a.href = url;
  a.download = `MOZI问卷_${new Date().toISOString().slice(0,10)}_${Date.now()}.json`;
  document.body.appendChild(a);
  a.click();
  document.body.removeChild(a);
  URL.revokeObjectURL(url);
  showToast('JSON 数据已导出', 'success');
}

function resetForm() {
  if (!confirm('确定要清空所有已填写的内容吗？此操作不可撤销。')) return;
  document.getElementById('surveyForm').reset();
  document.querySelectorAll('.option.checked').forEach(o => o.classList.remove('checked'));
  document.querySelectorAll('.option-other.visible').forEach(o => o.classList.remove('visible'));
  // 清空所有动态添加的输入行
  document.querySelectorAll('.list-input').forEach(list => {
    while (list.children.length > 1) list.removeChild(list.lastChild);
    const firstInput = list.querySelector('input');
    if (firstInput) firstInput.value = '';
  });
  document.querySelectorAll('.list-input').forEach(list => updateRemoveButtons(list.id));
  formData = {};
  updateProgress();
  showToast('已清空表单');
  window.scrollTo({ top: 0, behavior: 'smooth' });
}
